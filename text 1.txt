local function makeDraggable(handle, tgt)
	local dragging, start, orig, moved = false, nil, nil, false
	handle.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging, moved, start, orig = true, false, i.Position, tgt.Position
			i.Changed:Connect(function() if i.UserInputState == Enum.UserInputState.End then dragging = false end end)
		end
	end)
	conn(UIS.InputChanged, function(i)
		if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
			local d = i.Position - start
			if d.Magnitude > 4 then moved = true end
			tgt.Position = UDim2.new(orig.X.Scale, orig.X.Offset + d.X, orig.Y.Scale, orig.Y.Offset + d.Y)
		end
	end)
	return function() return moved end
end

local logBuf, consoleScroll, logN = {}, nil, 0
local function addLine(msg)
	logN += 1
	local l = new("TextLabel", {
		Size = UDim2.new(1, 0, 0, 15), BackgroundTransparency = 1, Text = ">> " .. msg:lower(), TextSize = 12, Font = Enum.Font.Code,
		TextColor3 = T.accent2, TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd, LayoutOrder = logN,
	}, consoleScroll)
	local kids = {}
	for _, c in ipairs(consoleScroll:GetChildren()) do if c:IsA("TextLabel") then kids[#kids + 1] = c end end
	if #kids > 60 then
		table.sort(kids, function(a, b) return a.LayoutOrder < b.LayoutOrder end)
		kids[1]:Destroy()
	end
	task.defer(function()
		if consoleScroll and consoleScroll.Parent then consoleScroll.CanvasPosition = Vector2.new(0, 1e5) end
	end)
end
local function log(msg)
	if consoleScroll then addLine(msg) else logBuf[#logBuf + 1] = msg end
end

local PD = {}
local function selectTab(name, instant)
	if view == name and not instant then
		if compact then setDrawer(false) end
		return
	end
	view = name
	for n, p in pairs(pages) do p.Visible = (n == name) end
	local p = pages[name]
	if not instant then
		p.Position = UDim2.fromOffset(0, 16)
		tw(p, { Position = UDim2.fromOffset(0, 0) }, 0.3, Enum.EasingStyle.Quint)
		UI.fadeO.BackgroundTransparency = 0.5
		tw(UI.fadeO, { BackgroundTransparency = 1 }, 0.28)
	end
	for n, t in pairs(tabs) do
		local on = n == name
		tw(t.btn, { BackgroundTransparency = on and 0 or 1 }, 0.2)
		tw(t.lbl, { TextColor3 = on and Color3.new(1, 1, 1) or T.dim }, 0.2)
		tw(t.bar, { BackgroundTransparency = on and 0 or 1 }, 0.2)
	end
	if compact then setDrawer(false) end
end

function UI.doSearch(q)
	q = (q or ""):lower()
	local firstHit, curHit
	for _, name in ipairs(tabOrder) do
		local d = PD[pages[name]]
		local any, cardAny = false, {}
		for _, e in ipairs(d.rows) do
			local card = e[3]
			local m = (q == "") or (e[2] ~= "" and e[2]:find(q, 1, true) ~= nil) or (card ~= nil and UI.titles[card] ~= nil and UI.titles[card]:find(q, 1, true) ~= nil)
			e[1].Visible = m and true or false
			if m then
				any = true
				if card then cardAny[card] = true end
			end
		end
		for _, c in ipairs(d.cards) do c.Visible = (q == "") or (cardAny[c] == true) end
		if q ~= "" and any then
			firstHit = firstHit or name
			if name == view then curHit = true end
		end
	end
	if q ~= "" and firstHit and not curHit then selectTab(firstHit) end
end

		local t = (S[key] - min) / (max - min)
		fill.Size = UDim2.fromScale(t, 1)
		knob.Position = UDim2.fromScale(t, 0.5)
		val.Text = tostring(math.floor(S[key] * 100 + 0.5) / 100)

		local v = math.clamp(math.floor((min + (max - min) * t) / step + 0.5) * step, min, max)
		if v ~= S[key] then
			S[key] = v
			show()
			if cb then cb(v) end
		end
	end
	hit.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			page.ScrollingEnabled = false
			setX(i.Position.X)
		end

		if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then setX(i.Position.X) end

X.GUNS = { "ak-47", "ak47", "m9", "m4", "remington", "revolver", "mp5", "fal", "shotgun", "pistol", "rifle", "gun" }

function X.role(plr)
	local n = plr.Team and plr.Team.Name:lower() or ""
	if n:find("guard", 1, true) then return "guard" end
	if n:find("criminal", 1, true) then return "criminal" end
	if n:find("inmate", 1, true) then return "inmate" end
	return nil
end
function X.teamColor(plr)
	if S.espTeamColor and plr.Team then
		local r = X.role(plr)
		if r == "guard" then return Color3.fromRGB(70, 140, 255) end
		if r == "inmate" then return Color3.fromRGB(255, 150, 40) end
		if r == "criminal" then return Color3.fromRGB(255, 70, 70) end
		return plr.TeamColor.Color
	end
	return Color3.new(1, 1, 1)
end
function X.isEnemy(plr)
	if plr.Team == nil or lp.Team == nil then return true end
	return plr.Team ~= lp.Team
end
function X.isGun(tool)
	if not (tool and tool:IsA("Tool")) then return false end
	if tool:FindFirstChild("GunStates", true) then return true end
	local n = tool.Name:lower()
	for _, g in ipairs(X.GUNS) do
		if n:find(g, 1, true) then return true end
	end
	return false
end
function X.heldGun(plr)
	local ch = (plr or lp).Character
	if not ch then return nil end
	for _, t in ipairs(ch:GetChildren()) do
		if X.isGun(t) then return t end
	end
	return nil
end
function X.hostile(plr)
	local role = X.role(plr)
	if role == "criminal" then return true end
	if role ~= "inmate" then return false end
	local ch = plr.Character
	local flag = plr:GetAttribute("Hostile") or plr:GetAttribute("hostile") or (ch and (ch:GetAttribute("Hostile") or ch:GetAttribute("hostile")))
	if flag then return true end
	local v = plr:FindFirstChild("Hostile") or (ch and ch:FindFirstChild("Hostile"))
	if v and v:IsA("BoolValue") and v.Value then return true end
	return S.aimInmateArmed and X.heldGun(plr) ~= nil
end
-- guards: criminals and hostile inmates only. inmates/criminals: nobody (unless "Prisoner side aims" is on)
function X.canTarget(plr)
	local me, her = X.role(lp), X.role(plr)
	if me == "guard" then
		if her == "criminal" then return true end
		return her == "inmate" and X.hostile(plr)
	end
	if me == "inmate" or me == "criminal" then
		return S.aimPrisoner and her == "guard"
	end
	return false
end

function X.trDrop(plr)
	local t = X.tr[plr]

	pcall(function() t.beam:Destroy() end)

function X.trClear()

end
function X.espDrop(plr)
	local r = X.esp[plr]
	X.trDrop(plr)
	if not r then return end
	for _, k in ipairs({ "hl", "bb" }) do
		if r[k] then pcall(function() r[k]:Destroy() end) end
	end
	X.esp[plr] = nil
end
function X.espMake(plr)
	local ch = plr.Character
	local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
	local hum = ch and ch:FindFirstChildOfClass("Humanoid")
	if not (hrp and hum) then return nil end
	local r = X.esp[plr]
	if r and r.ch == ch then return r end
	X.espDrop(plr)
	r = { ch = ch, hrp = hrp, hum = hum }
	r.hl = new("Highlight", { Adornee = ch, FillTransparency = 0.6, OutlineTransparency = 0, DepthMode = Enum.HighlightDepthMode.AlwaysOnTop }, ch)
	r.bb = new("BillboardGui", { Adornee = hrp, Size = UDim2.fromOffset(200, 22), StudsOffset = Vector3.new(0, 3.4, 0), AlwaysOnTop = true }, hrp)
	r.lbl = new("TextLabel", { Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = plr.DisplayName, TextColor3 = Color3.new(1, 1, 1), TextStrokeTransparency = 0.3, Font = Enum.Font.GothamBold, TextSize = S.espSize }, r.bb)
	X.esp[plr] = r
	return r
end
function X.espUpdate(rt)
	if not S.esp then
		for plr in pairs(X.esp) do X.espDrop(plr) end
		return
	end
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= lp then
			local show = not (S.espEnemyOnly and not X.isEnemy(plr))
			local r = show and X.espMake(plr)
			if r then
				local d = rt and (r.hrp.Position - rt.Position).Magnitude or 0
				local on = d <= S.espMax and r.hum.Health > 0
				r.hl.Enabled, r.bb.Enabled = on, on
				if on then
					local col = X.teamColor(plr)
					r.hl.FillColor, r.hl.OutlineColor, r.lbl.TextColor3, r.lbl.TextSize = col, col, col, S.espSize
					local parts = {}
					if S.espName then parts[#parts + 1] = plr.DisplayName end
					if S.espHealth then parts[#parts + 1] = math.floor(r.hum.Health) .. "HP" end
					if S.espDist then parts[#parts + 1] = math.floor(d) .. "m" end
					r.lbl.Text = table.concat(parts, "  ")
				end
			elseif not show then
				X.espDrop(plr)
			end
		end
	end
	for plr in pairs(X.esp) do
		if not plr.Parent then X.espDrop(plr) end
	end
end
function X.trUpdate(rt)
	if not (S.tracers and S.esp) or not rt then
		if next(X.tr) ~= nil or X.a0 then X.trClear() end
		return
	end
	if not X.a0 or X.a0.Parent ~= rt then
		if X.a0 then pcall(function() X.a0:Destroy() end) end
		X.a0 = new("Attachment", {}, rt)
	end
	for plr, r in pairs(X.esp) do
		local t = X.tr[plr]
		if r.bb.Enabled then
			if not t then

				local beam = new("Beam", { Attachment0 = X.a0, Attachment1 = att, Width0 = 0.1, Width1 = 0.1, FaceCamera = true, LightEmission = 1, Transparency = NumberSequence.new(0.2) }, r.hrp)
				t = { beam = beam, att = att }
				X.tr[plr] = t
			end
			t.beam.Color = ColorSequence.new(X.teamColor(plr))
		elseif t then
			X.trDrop(plr)
		end
	end
end

-- ===== hitbox expander =====
function X.hitRestore(part)
	local o = X.hit[part]
	if not o then return end
	pcall(function()
		part.Size, part.Transparency, part.CanCollide = o[1], o[2], o[3]
	end)
	X.hit[part] = nil
end
function X.hitRestoreAll()
	for part in pairs(X.hit) do X.hitRestore(part) end
end
function X.hitStep()
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= lp then
			local hrp = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
			if hrp then
				local want = S.hitbox and (not S.hitboxTeam or X.canTarget(plr))
				if want then
					if not X.hit[hrp] then X.hit[hrp] = { hrp.Size, hrp.Transparency, hrp.CanCollide } end
					local s = S.hitboxSize
					hrp.Size = Vector3.new(s, s, s)
					hrp.Transparency = 0.75
					hrp.CanCollide = false
				elseif X.hit[hrp] then
					X.hitRestore(hrp)
				end
			end
		end
	end
end

function X.visible(cam, part, ch)
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { lp.Character, ch }
	params.IgnoreWater = true
	local o = cam.CFrame.Position
	return workspace:Raycast(o, part.Position - o, params) == nil
end
function X.getTarget(cam, fov)
	fov = fov or S.aimFov
	local best, bd
	local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
	local myRoot = getRoot()
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= lp and X.canTarget(plr) then
			local ch = plr.Character
			local hum = ch and ch:FindFirstChildOfClass("Humanoid")
			local part = ch and (ch:FindFirstChild(S.aimPart) or ch:FindFirstChild("HumanoidRootPart"))
			if hum and part and hum.Health > 0 and not ch:FindFirstChildOfClass("ForceField") then
				local pos, on = cam:WorldToViewportPoint(part.Position)
				if on then
					local d2 = (Vector2.new(pos.X, pos.Y) - center).Magnitude
					local dist = myRoot and (part.Position - myRoot.Position).Magnitude or 0
					if d2 <= fov and dist <= S.aimDist and (not bd or d2 < bd) then
						if (not S.aimWall) or X.visible(cam, part, ch) then best, bd = part, d2 end
					end
				end
			end
		end
	end
	return best, bd
end
function X.shoot(gun)
	if S.fireMode == "click" and mouse1click then
		pcall(mouse1click)
	else
		pcall(function() gun:Activate() end)
	end
end

		end
	end

function X.giveItem(name)
	task.spawn(function()
		local ok, err = pcall(function()
			local items = workspace:FindFirstChild("Prison_ITEMS")
			local giver = items and items:FindFirstChild("giver")

			local remote = workspace:FindFirstChild("Remote")
			local ih = remote and remote:FindFirstChild("ItemHandler")
			if not part then error(name .. " not found in the map") end
			if not ih then error("ItemHandler remote not found") end
			ih:InvokeServer(part)
		end)
		if ok then notify("Item", "Asked for " .. name, T.green, 2.5) else notify("Item", tostring(err), T.red, 4) end
	end)
end
function X.setTeam(colorName)

		ev:FireServer(colorName)
	end)

end
function X.scan()
	local function dump(title, folder)
		if not folder then log(title .. ": not found") return end
		local names = {}
		for _, c in ipairs(folder:GetChildren()) do names[#names + 1] = c.Name end
		table.sort(names)
		log(title .. ": " .. #names .. " found")
		local chunk = {}
		for i, n in ipairs(names) do
			chunk[#chunk + 1] = n
			if #chunk == 4 or i == #names then
				log(table.concat(chunk, ", "))
				chunk = {}
			end
		end
	end
	dump("workspace.Remote", workspace:FindFirstChild("Remote"))
	dump("Prison_ITEMS.giver", workspace:FindFirstChild("Prison_ITEMS") and workspace.Prison_ITEMS:FindFirstChild("giver"))
end
function X.serverHop()
	task.spawn(function()
		local ok, res = pcall(function()
			return HttpService:JSONDecode(game:HttpGet("https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"))
		end)
		if not ok or type(res) ~= "table" or not res.data then notify("Server hop", "Could not list servers", T.red, 3) return end
		local list = {}
		for _, s in ipairs(res.data) do
			if s.id ~= game.JobId and s.playing < s.maxPlayers then list[#list + 1] = s.id end
		end
		if #list == 0 then notify("Server hop", "No free server found", T.yellow, 3) return end
		notify("Server hop", "Joining another server...", T.accent2, 3)
		pcall(function() TPS:TeleportToPlaceInstance(game.PlaceId, list[math.random(#list)], lp) end)
	end)
end

function X.unload()
	if not alive then return end
	alive = false
	for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
	pcall(function() for plr in pairs(X.esp) do X.espDrop(plr) end end)
	pcall(X.trClear)

	pcall(function()
		local h = getHum()
		if h then h.WalkSpeed = 16 h.PlatformStand = false end
		for k, v in pairs(L0) do Lighting[k] = v end
	end)
	pcall(function() if UI.blur then UI.blur:Destroy() end end)
	pcall(saveConfig)
	pcall(function() X.circle:Destroy() end)
	pcall(function() gui:Destroy() end)
	G.HappyPLUnload = nil
end

	pressFx(floatBtn)
	local wasMoved = makeDraggable(floatBtn, floatBtn)
	floatBtn.MouseButton1Click:Connect(function()
		if wasMoved() then return end
		UI.main.Visible = not UI.main.Visible
	end)
end
UI.closeB.MouseButton1Click:Connect(function() X.unload() end)
UI.minB.MouseButton1Click:Connect(function() UI.main.Visible = false end)

-- ===== events =====
local function bindChar(char)
	local hum = char:WaitForChild("Humanoid", 10)
	if hum then
		conn(hum.Died, function() X.deathPos = X.lastPos end)
	end
end
if lp.Character then task.spawn(bindChar, lp.Character) end
conn(lp.CharacterAdded, function(c)
	task.spawn(bindChar, c)
	X.burst = 0
	if S.autoRespawn and X.deathPos then
		local p = X.deathPos
		task.wait(1.2)
		local rt = getRoot()
		if rt and alive then rt.CFrame = CFrame.new(p + Vector3.new(0, 3, 0)) end
		X.deathPos = nil
	end
end)
conn(lp.Idled, function()
	if S.afk then pcall(function() VU:CaptureController() VU:ClickButton2(Vector2.new()) end) end
end)
conn(Players.PlayerRemoving, function(plr) X.espDrop(plr) end)
conn(UIS.JumpRequest, function()
	if S.infJump then
		local h = getHum()
		if h then pcall(function() h:ChangeState(Enum.HumanoidStateType.Jumping) end) end
	end
end)

-- ===== main loop =====
local errOnce = false
conn(RunService.Heartbeat, function()
	if not alive then return end
	local ok, err = pcall(function()
		local now = os.clock()
		local rt, hum = getRoot(), getHum()
		local cam = workspace.CurrentCamera

		X.frames += 1
		if now - X.fpsT >= 0.5 then
			X.fps = math.floor(X.frames / (now - X.fpsT) + 0.5)
			X.frames, X.fpsT = 0, now
		end
		floatBtn.Visible = S.floatBtn or not UI.main.Visible

		if now - X.tPerf > 0.5 then
			X.tPerf = now
			local ping = 0
			pcall(function() ping = math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue()) end)
			UI.perfL.Text = X.fps .. " FPS  |  " .. ping .. " ms"
			UI.perfL.Visible = S.hud
			local alv = hum and hum.Health > 0
			UI.statusL.Text = 'STATUS: <font color="' .. (alv and "#5ae68c" or "#ff5660") .. '">' .. (alv and "READY" or "DEAD") .. "</font>"
			X.info.Team.Text = lp.Team and lp.Team.Name or "None"
			X.info.Players.Text = tostring(#Players:GetPlayers())
			X.info.Status.Text = alv and "Ready" or "Dead"
			if logoId and not X.logoApplied then
				X.logoApplied = true
				for _, il in ipairs(X.logoTargets) do il.Image = logoId end
			end
		end

		if now - X.tEsp > 0.1 then
			X.tEsp = now
			X.espUpdate(rt)
			X.trUpdate(rt)
		end
		if now - X.tHit > 0.4 and (S.hitbox or next(X.hit) ~= nil) then
			X.tHit = now
			X.hitStep()
		end

		if not rt or not hum then return end
		local live = hum.Health > 0
		if live and now - X.tPos > 1 and hum.FloorMaterial ~= Enum.Material.Air then
			X.tPos = now
			X.lastPos = rt.Position
		end

		if S.speed and live then hum.WalkSpeed = S.speedVal end
		if S.jump and live then
			hum.UseJumpPower = true
			hum.JumpPower = S.jumpVal
		end
		if S.fly and cam and live then
			hum.PlatformStand = true
			local cf = cam.CFrame
			local rel = cf:VectorToObjectSpace(hum.MoveDirection)
			local dir = cf.LookVector * (-rel.Z) + cf.RightVector * rel.X
			if UIS:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.new(0, 1, 0) end
			if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then dir -= Vector3.new(0, 1, 0) end
			rt.AssemblyLinearVelocity = dir * S.flySpeed
		end
		if S.noclip or S.fly then
			local ch = lp.Character
			if ch then
				for _, d in ipairs(ch:GetDescendants()) do
					if d:IsA("BasePart") and d.CanCollide then d.CanCollide = false end
				end
			end
		end
		if S.fullbright then
			Lighting.Brightness, Lighting.ClockTime, Lighting.GlobalShadows = 2, 14, false
			Lighting.Ambient, Lighting.OutdoorAmbient = Color3.fromRGB(190, 190, 190), Color3.fromRGB(190, 190, 190)
		end
		if now - X.tSave > 30 then
			X.tSave = now
			saveConfig()
		end
	end)
	if not ok and not errOnce then
		errOnce = true
		report("loop", err)
	end
end)
